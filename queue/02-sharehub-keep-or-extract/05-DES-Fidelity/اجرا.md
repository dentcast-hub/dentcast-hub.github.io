```bash
K=/home/user/dentcast-hub.github.io/.dentcast/des-v3.5-knowledge.md

claude -p --tools "" --strict-mcp-config --no-session-persistence --system-prompt-file "$K" --model opus --effort high --output-format json < des-runs/input-source-S1.json > des-runs/out-source-S1.envelope.json

python3 -I tools/des_fidelity_units.py sharehub/share-QUEUE02 --texts ../../out-des/source_text.json --build ../../out-des/fidelity-runs/run1
# checker runs 1 and 2, in parallel, identical input, separate processes
claude -p --tools "" --strict-mcp-config --no-session-persistence --system-prompt-file "$K" --model sonnet --effort medium --output-format json < run1/input-src0.json > run1/receipts/out-src0-{1,2}.envelope.json
python3 -I tools/des_fidelity_units.py sharehub/share-QUEUE02 --vote run1        # -> tie-break u9,u14,u47,u64,u99; review u38
python3 -I tools/des_fidelity_units.py sharehub/share-QUEUE02 --texts source_text.json --build run1-tiebreak --only u9,u14,u47,u64,u99
claude -p … --model sonnet --effort medium … < run1-tiebreak/input-src0.json   # -> out-src0-3.json (copied into run1/)
claude -p … --model opus --effort high … < run1/receipts/review/input-review.json   # -> run1/review.json
python3 -I tools/des_fidelity_units.py sharehub/share-QUEUE02 --vote run1        # exit 0
python3 -I tools/des_fidelity_units.py sharehub/share-QUEUE02 --orphans run1 --texts source_text.json   # 14 orphans
claude -p … --model opus --effort high … < run1/receipts/orphans/input-orphans.json   # -> run1/orphans.json
python3 -I tools/des_fidelity_units.py sharehub/share-QUEUE02 --sweep run1       # found u30,u36,u64,u103; 10 in no source
python3 -I tools/des_fidelity_units.py sharehub/share-QUEUE02 --texts source_text.json --build run2-orphan-recheck --only u30,u36,u64,u103
claude -p … --model sonnet --effort medium … < run2-orphan-recheck/input-src0.json   # x2 parallel
python3 -I tools/des_fidelity_units.py sharehub/share-QUEUE02 --texts source_text.json --build run2-orphan-recheck-tiebreak --only u36,u103
claude -p … --model sonnet --effort medium … < run2-orphan-recheck-tiebreak/input-src0.json   # -> out-src0-3.json (copied into run2-orphan-recheck/)
python3 -I tools/des_fidelity_units.py sharehub/share-QUEUE02 --vote run2-orphan-recheck --base run1   # exit 0 -> voted-src0.json = fidelity.json
```
